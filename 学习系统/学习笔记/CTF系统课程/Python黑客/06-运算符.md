### 概览
#### 基础运算符

| 运算符 | 描述  | 示例          |
| --- | --- | ----------- |
| +   | 加   | a = 10 + 1  |
| -   | 减   | b = 10 - 1  |
| *   | 乘   | c = 10 * 1  |
| /   | 除   | d = 10 / 1  |
| //  | 整除  | e = 10 // 1 |
| %   | 取模  | f = 10 % 1  |
| **  | 幂   | g = 10 ** 1 |
#### 代码示例
```python
## 列表合并  
list1 = [1,2,3]  
list2 = [4,5,6]  
combined_list = list1 + list2  
print(combined_list)  
  
# #数字减法  
x= 10  
y = 2  
result = x - y  
print(result)  
  
#集合差集  
set1 = {1,2,3,4}  
set2 = {3,4,5}  
difference = set1 - set2  
print(difference)  
  
#数字乘法  
x = 10  
y = 5  
result = x * y  
print(result)  
  
#字符串重复  
str1 = "python"  
repeated_str = str1 * 3    #结果为pythonpythonpython  
print(repeated_str)  
  
#列表重复  
list2 = [1,2]  
repeated_list2 = list2 * 3 #结果为[1, 2, 1, 2, 1, 2]  
print(repeated_list2)  
  
#数字除法  
x = 10  
y = 5  
result = x / y  #结果为2.0  
print(result)  
  
#整数除法  
x = 10  
y = 5  
result = x // y #结果为2  
print(result)  
  
#负数整除  
x = -10  
y = 5  
result = x // y #结果为-2  
print(result)  
  
#计算余数  
x = 10  
y = 5  
result = x % y  #结果为0  
print(result)  
  
#循环索引  
 for i in range(5):  
     print(i % 3)  #输出为01201，将1-3循环5次  
  
#计算平方  
x = 5  
result = x ** 2 #结果为25  
print(result)  

#计算立方  
 x = 5  
 result = x ** 3 #结果为125  
 print(result)  
  
#计算任意幂  
 x = 2  
 y = 8  
 result = x ** y  #结果为256  
 print(result)
```
#### 比较运算符

| 运算符 | 描述   | 示例     |
| --- | ---- | ------ |
| ==  | 等于   | a == b |
| !=  | 不等于  | a !=   |
| >   | 大于   | a > b  |
| <   | 小于   | a < b  |
| >=  | 大于等于 | a >= b |
| <=  | 小于等于 | a <= b |
```python
#数字比较

x = 5

y = 5.0

result = x == y   #结果为True

print(result)

  

#字符串比较

str1 = "python"

str2 = "python"

result = str1 == str2   #结果为True

print(result)

  

#列表比较

list1 = [1,2,3,4,5]

list2 = [1,2,3,4,5]

list3 = list1 == list2   #结果为True

print(list3)

  

# 大于运算符

# 数字比较

x = 5

y = 3

result = x > y   #结果为True

print(result)

  

# 字符串比较

str1 = "python"

str2 = "java"

result = str1 > str2

print(result)  #结果为True，因为"p"的ASCII码大于"j"

  

# 小于运算符

# 数字比较

x = 2

y = 6

result = x < y   #结果为True

print(result)

  

# # 字符串比较

str1 = "python"

str2 = "java"

result = str1 < str2

print(result)  #结果为False，因为"p"的ASCII码大于"j"

```
#### 赋值运算符！！！


| 符号  | 描述    | 示例      |
| --- | ----- | ------- |
| =   | 赋值    | a = 5   |
| +=  | 加后赋值  | a += 5  |
| -=  | 减后符号  | a -= 5  |
| *=  | 乘后赋值  | a *= 5  |
| /=  | 除后赋值  | a /= 5  |
| //= | 整除后赋值 | a //= 5 |
| %=  | 取模后赋值 | a %= 5  |
| **= | 幂后赋值  | a **= 5 |
```python
# 赋值运算符！！！

# 简单赋值

x = 10

print(x)

  

# 复合赋值

x += 5  # 相当于 x = x + 5

print(x)

  

# 数字加法赋值

x = 4

x += 10  # 结果为14

print(x)

  

# 字符串拼接赋值

str1 = "Hello, "

str1 += "World!"  # 结果为"Hello, World!"

print(str1)

  
  

# 减法赋值运算符

x = 10

x -= 5 # 结果为5

print(x)

  

# 数字乘法赋值

x = 5

x *= 3  # 结果为15

print(x)

  

# 数字除法赋值

x = 20

x /= 2  # 结果为10.0

print(x)

  
  

# 数字整除赋值

x = 20

x //= 3  # 结果为6

print(x)  

  

# 数字取模赋值

x = 10

x %= 2  # 结果为0

print(x)

  
  

# 幂运算赋值

x = 100

x **= 10 # 结果为10000000000

print(x)
```

# 蓝队平台-监控系统

引言

基于红队的搭建基础，从搭建到系统设置再到登录，与红队一致！！！

### 系统更新

#### 基础环境搭建
    
    
    #	1.系统更新
    sudo apt update && sudo apt upgrade -y
    
    #	2.安装核心工具
    sudo apt install -y curl wget git vim tmux htop tree
    
    #	3.配置防火墙 拒绝流量那个也可以做完，单独划几个网段让自家队友访问
    sudo ufw enable
    sudo ufw default deny incoming  	#一个人玩的话，就拒绝所有流量访问，团队的话就不需要输入这个
    sudo ufw default allow outgoing
    
    #	开放ssh（管理）
    sudo ufw allow ssh
    
    #	开放监控平台端口
    sudo ufw allow 80/tcp    # HTTP访问
    sudo ufw allow 443/tcp   # HTTPS访问
    sudo ufw allow 9000/tcp  # Portainer
    sudo ufw allow 9090/tcp  # Cockpit
    sudo ufw allow 19999/tcp # NetData
    sudo ufw allow 3000/tcp  # Grafana
    sudo ufw allow 8000/tcp  # 文件共享
    sudo ufw allow 5230/tcp  # Memos笔记
    
    #	限制访问来源（重要！！！）
    #	方法一
    sudo ufw allow from 192.168.10.0/24 to any port 80,443,9000,19999,3000/tcp
    
    #	方法二
    sudo ufw allow from 192.168.10.0/24 to any port 80/tcp	#依次类推
    
    #	如果不小心加入了一些敏感端口或者规则，就用这个
    #	删除不需要的规则
    #	删除对任何人开放的端口
    sudo ufw delete allow 9090/tcp
    sudo ufw delete allow 8000/tcp
    #	同时删除IPv6的对应端口，这个方法有点慢，不建议
    sudo ufw delete allow 9090/tcp from anywhere to anywhere comment 'v6'
    sudo ufw delete allow 8000/tcp from anywhere to anywhere comment 'v6'
    ...
    
    #	或者用编号删除的方法，推荐！！！
    #	先查看编号
    sudo ufw status numbered
    
    #	根据编号删除！！！
    sudo ufw delete 10  # 6379/tcp
    sudo ufw delete 8   # 8080/tcp  
    sudo ufw delete 5   # 9090/tcp
    sudo ufw delete 7   # 8000/tcp
    sudo ufw delete 9   # 5320/tcp
    
    #	
    
    
    
    #	启用防火墙
    sudo ufw --force enable

### 配置核心服务

#### 安装docker
    
    
    # 安装Docker和Docker Compose
    sudo apt install -y docker.io docker-compose
    
    # 启动并启用Docker服务
    sudo systemctl enable docker
    sudo systemctl start docker
    
    # 将用户添加到docker组
    sudo usermod -aG docker $USER
    
    # 重新登录或使用newgrp
    newgrp docker

#### 验证docker安装
    
    
    # 检查Docker版本
    docker --version
    
    # 检查Docker Compose版本  
    docker-compose --version
    
    # 查看Docker服务状态
    sudo systemctl status docker
    
    # 运行测试容器
    docker run hello-world			#	如果这一步报错或者没下载下来，就该docker的镜像地址，阿里云申请一个就行

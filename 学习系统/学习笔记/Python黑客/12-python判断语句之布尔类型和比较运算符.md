比较运算符的用法
```python
# 布尔类型  
bool_1 = True  
bool_2 = False  
print(f"bool_1变量的内容是：{bool_1},类型是：{type(bool_1)}")  
print(f"bool_2的变量内容是：{bool_2},类型是：{type(bool_2)}")  
  
# 比较运算符的用法  
# == != >= <= > <  
num1 = 10  
num2 = 10  
print(f"10 == 10 的解雇偶是：{num1 == num2}")  
  
num1 = 10  
num2 = 15  
print(f"10 != 15的结果是：{num1 != num2}")  
  
name1 = "itpython"  
name2 = "iloveyou"  
print(f"itpython == iliveyou的结果是：{name1 == name2}")  
  
num1 = 10  
num2 = 5  
print(f"10 > 5的结果是：{num1 > num2}")  
print(f"10 < 5的结果是：P{num1 < num2}")  
  
num1 = 10  
num2 = 10  
print(f"10 >= 5的结果是：{num1 >= num2}")  
print(f"10 <= 5的结果是:{num1 <= num2}")
```
